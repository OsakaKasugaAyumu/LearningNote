
**最小CMakeLists.txt**
~~~
/*
    cmake_mininum_required : cmake内置命令
    cmake_mininum_required(VERSION 3.20.0), 最低cmake版本需求, 如果cmake版本低于3.20.0直接报错
*/
cmake_mininum_required(VERSION 3.20.0)


/*
    find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE}) : cmake命令, 用来寻找并加载外部项目/库
        - Zephyr : 包名
        - REQUIRED : 强制要求, 如果找不到Zephyr, 直接报错
        - HINTS : 提示词, 高速CMAKE去哪里找
        - $ENV{ZEPHYR_BASE} : 读取系统环境变量ZEPHYR_BASE

    这个命令为C语言直接使用 #include <zephyr/kernel.h>提供环境支持
*/
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})


/*
    project(my_app) : cmake命令
    工程名称, 初始化项目
*/
project(my_app)


/*
    target_sources(app PRIVATE src/main.c) : cmake命令, 用于给指定的"目标"添加源文件
        - app : 可执行文件目标
        - PRIVATE : 作用域限制
        - src/main.c : C语言源码路径
*/
target_sources(app PRIVATE src/main.c)

~~~



**标准CMakeLists.txt**
~~~

/*
    project(my_sensor_app VERSION 1.0.0)
        - PROJECT_VERSION_MAJOR (1)
        - PROJECT_VERSION_MINOR (0)
        - PROJECT_VERSION_PATCH (0)
*/
cmake_mininum_required(VRESION 3.20.0)
find_package(Zephyr REQUIRED $ENV{ZEPHYR_BASE})
project(my_sensor_app VERSION 1.0.0)
target_sources(app PRIVATE src/main.c src/sensor.c src/display.c)


/*
    target_include_directories(app PRIVATE include) : 把当前目录下的include目录加入头文件搜索路径
        - 给GCC编译器加上了 -I include
*/
target_include_directories(app PRIVATE include)


/*
    target_compile_options(app PRIVATE -Wall -Wextra) : 把后面参数传递给底层GCC编译器
        - -Wall : 启用所有的警告, 变量定义但是没使用
        - -Wextra : 启用额外警告
*/
target_compile_options(app PRIVATE -Wall -Wextra)


/*
    target_compile_definitions(app PRIVATE APP_VERSION_MAJOR=${PROJECT_VERSION_MAJOR} APP_VERSION_MINOR=${PROJECT_VERSION_MINOR}) : 编译时, 向C代码注入全局宏
        - 相当于给编译器加了-DAPP_VERSION_MAJOR=1 -DAPP_VERSION_MINOR=0
        - 实现了CMake变量到C语言宏的完美同步
        - 在C语言直接使用 printf("%d/%d", APP_VERSION_MAJOR, APP_VERSION_MINOR);
*/
target_compile_definitions(app PRIVATE APP_VERSION_MAJOR=${PROJECT_VERSION_MAJOR} APP_VERSION_MINOR=${PROJECT_VERSION_MINOR})


/*
    和prj.conf / Kconfig配合使用
*/
if(CONFIG_DEBUG)
    target_compile_definitions(app PRIVATE DEBUG_MODE=1)
endif()

/*
    链接自定义库
*/
target_link_libraries(app PRIVATE my_custom_lib)
~~~
